menu status {
  Global News Script
  .globalNews.ini
  ..Enable:/enable #globalNews
  ..Disable:/disable #globalNews
}
;
alias globalNews { 
Enable #globalNews |
echo -a $time |
echo -a $asctime(HH:nn:ss tt) |
echo -a $date |
echo -a $asctime(dddd) |
echo -a $asctime(mmmm) |
echo -a $date(yyyy) |
set %joursParAnneeNonBissextile 365 |
set %joursParAnneeBissextile 366 |
set %anneeEntreAnneeBissextile 4 |
set %derniereAnneeBissextile 2024 |
set %prochaineAnneeBissextile 2028 |
}

    msg $chan 4,0.:.1 Syntaxe: !setnews n° news 4,0.:. 
    halt 
  } 
  if (!$3) { 
    msg $chan 4,0.:.1 Syntaxe: !setnews n° news 4,0.:. 
    halt 
  } 
  if ($2 !isnum) { 
    msg $chan 4,0.:.1 Syntaxe: !setnews n° news 4,0.:. 
    halt 
  }

    msg $chan 4,0.:.1 Le n° doit être comprit entre 0 & 1000 4,0.:.
